
# Audio2Midi
Python script to turn MP3 into MIDI

# Usage

	python main.py input.mp3 -o output.mid
# Requirements

 - ffmpeg
 - pydub
 - numpy
 - scipy
 - mido
 - numba
 - audioop-lts
# Installation (I'm using uv but you can use whatever you want

    git clone https://github.com/cylin577Audio2Midi
    cd Audio2Midi
    uv venv
	source .venv/bin/activate
	uv pip install -r requirements.txt
